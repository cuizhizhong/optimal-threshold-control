由推论\ref{cor:unique-nontrivial-root}，非平凡根 $\sigma$ 必落在
$\Theta(s,i(s))$ 沿特征线的严格下降段上。将\eqref{eq:switch-param}第二式
的左端视为自由变量 $s$ 的函数，即令
\begin{equation}
F(s;\bar s)
=\Theta\big(s,\;\Psi(\bar s)+\ell\ln s\big)-\Theta(\bar s,\bar i),
\qquad
\Psi(\bar s)=a(\bar s)-\ell\ln\bar s,
\label{eq:switch-residual}
\end{equation}
则
\[
\frac{\partial F}{\partial s}\bigg|_{(\sigma,\bar s)}
=\frac{\ell\,[\,i_\sigma-g(\sigma)\,]}{i_\sigma^{\,2}(\sigma-r)^{2}}<0 .
\]
由隐函数定理，$\sigma$ 在其定义区间内部是 $\bar s$ 的 $C^{1}$ 函数。以下即可
直接对\eqref{eq:switch-param}求导。

\begin{lemma}[切换曲线的单调参数化]
\label{lem:sigma-monotone}
在 $\sigma$ 有定义的开区间内部，
\[
\sigma'(\bar s)<0,
\qquad
i_\sigma'(\bar s)<0 .
\]
因此 $\Gamma$ 是相平面中一条 $i$ 随 $s$ 严格递增的简单弧，可写成函数图像
$i=\gamma(s)$。
\end{lemma}

\begin{proof}
记 $\bar\Theta(\bar s)=\Theta(\bar s,a(\bar s))$。由\eqref{eq:Theta}，
\[
\Theta_s=\frac{\ell}{i(s-r)^{2}},
\qquad
\Theta_i=-\frac{\Theta}{i}.
\]

\emph{第 1 步（求导）.}
沿 $q=1$ 特征线 $\dd i/\dd s=\ell/s$，沿安全边界 $a'(s)=(h-s)/s$，故
\begin{equation}
D(s)=\Theta_s+\frac{\ell}{s}\Theta_i
=\frac{\ell\,[\,i-g(s)\,]}{i^{2}(s-r)^{2}},
\qquad
E(s)=\Theta_s+a'(s)\,\Theta_i
=\frac{\ell\,s\,i+(s-h)^{2}(s-r)}{s\,i^{2}(s-r)^{2}}>0 .
\label{eq:DE}
\end{equation}
$E>0$ 表明 $\Theta$ 沿安全边界严格递增。又
$\Psi'(\bar s)=a'(\bar s)-\ell/\bar s=(r-\bar s)/\bar s$，而切换点
$P=(\sigma,\;\Psi(\bar s)+\ell\ln\sigma)$ 满足
\[
\frac{\dd P}{\dd\bar s}
=\sigma'\Big(1,\frac{\ell}{\sigma}\Big)+\Psi'(\bar s)\,(0,1).
\]
对\eqref{eq:switch-param}第二式关于 $\bar s$ 求导，并用切换条件
$\Theta(P)=\bar\Theta$ 及 $\Theta_i(P)=-\bar\Theta/i_\sigma$，得
\begin{equation}
\sigma'(\bar s)\,D(\sigma)
=E(\bar s)-\frac{(\bar s-r)\,\bar\Theta}{\bar s\,i_\sigma}.
\label{eq:sigma-prime}
\end{equation}
由第一段所述 $D(\sigma)<0$，故只需证\eqref{eq:sigma-prime}右端为正。

\emph{第 2 步（化简判据）.}
代入 $\bar\Theta=(\bar s-h)/[\bar i(\bar s-r)]$，右端不等式
$E(\bar s)>(\bar s-r)\bar\Theta/(\bar s\,i_\sigma)$ 化为
\[
\frac{\ell\,\bar s\,\bar i+(\bar s-h)^{2}(\bar s-r)}
     {\bar s\,\bar i^{\,2}(\bar s-r)^{2}}
>\frac{\bar s-h}{\bar s\,\bar i\,i_\sigma}.
\]
两边同乘正量 $\bar s\,\bar i^{\,2}(\bar s-r)^{2}$：
\[
\ell\,\bar s\,\bar i+(\bar s-h)^{2}(\bar s-r)
>\frac{(\bar s-h)\,\bar i\,(\bar s-r)^{2}}{i_\sigma}.
\]
由\eqref{eq:switch-param}第二式解出
$i_\sigma=\bar i(\bar s-r)(\sigma-h)/[(\bar s-h)(\sigma-r)]$，代入右端并约去
$\bar i$ 与一个 $(\bar s-r)$：
\[
\ell\,\bar s\,\bar i+A
>\frac{A\,(\sigma-r)}{\sigma-h},
\qquad
A:=(\bar s-h)^{2}(\bar s-r)>0 .
\]
两边乘 $(\sigma-h)>0$ 并移项，注意 $(\sigma-r)-(\sigma-h)=h-r=\ell$，两边的
$\ell$ 相消，得等价形式
\begin{equation}
\sigma'(\bar s)<0
\iff
g(\bar s)\,(\bar s-h)<\bar i\,(\sigma-h).
\label{eq:sigma-criterion}
\end{equation}

\emph{第 3 步（验证判据）.}
非平凡根的存在蕴含 $\bar i>g(\bar s)$（否则由\eqref{eq:g-minus-i}，
$i-g$ 在 $[\bar s,\infty)$ 上恒负，$\Theta$ 严格递减，无非平凡根）；又
$\sigma>\bar s>h$ 给出 $\sigma-h>\bar s-h>0$。两式逐项相乘即得
\eqref{eq:sigma-criterion}的右端，故 $\sigma'(\bar s)<0$。

\emph{第 4 步（$i_\sigma$）.}
由\eqref{eq:switch-param}第一式 $i_\sigma=\Psi(\bar s)+\ell\ln\sigma$，
\[
i_\sigma'(\bar s)=\Psi'(\bar s)+\frac{\ell\,\sigma'(\bar s)}{\sigma}
=\frac{r-\bar s}{\bar s}+\frac{\ell\,\sigma'(\bar s)}{\sigma}<0,
\]
两项均为负（第一项因 $\bar s\ge h>r$）。
\end{proof}

\begin{remark}
引理\ref{lem:sigma-monotone}只在开区间内部成立。设 $s_e$ 为
$a(s_e)=g(s_e)$ 的根，即安全边界与曲线 $\{i=g(s)\}$ 的交点。当
$\bar s\to s_e$ 时 $\bar i\to g(\bar s)$ 且 $\sigma\to\bar s$，
\eqref{eq:sigma-criterion}中的两个不等式同时退化为等式，
\eqref{eq:sigma-prime}的两端同时趋于零，隐函数定理在该端点失效。
这对应于 $\Gamma$ 在 $s_e$ 处以退化方式并入安全边界。
\end{remark}

\begin{remark}
引理\ref{lem:sigma-monotone}使假设 (G2) 中" $\Gamma$ 为简单分段 $C^1$ 曲线"
不再需要单独假设：$C^1$ 性来自\eqref{eq:switch-residual}后的隐函数定理论证，
简单性来自 $\sigma$ 的严格单调。此外 $\Gamma$ 可写成图像 $i=\gamma(s)$，
这为第\ref{sec:synthesis}节中"状态位于 $\Gamma$ 的哪一侧"的判定提供了
直接依据。
\end{remark}
